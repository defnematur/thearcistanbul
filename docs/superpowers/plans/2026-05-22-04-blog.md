# Blog Implementation Plan (Index, Detail, MDX, Spotify, FTS Search)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Public-facing blog. Listing at `/[locale]/blog`, detail at `/[locale]/blog/[slug]`, MDX rendering with allowlisted components, Spotify embed at the top, hreflang TR↔EN linking, missing-translation fallback, and a Postgres full-text search across published posts.

**Architecture:** All blog pages are Server Components with ISR (`revalidate = 60`). DB queries live in `lib/db/queries/posts.ts` and are deduplicated per-request with `React.cache()`. Markdown is rendered via `next-mdx-remote/rsc` with a fixed component allowlist (no `rehype-raw`). The Spotify URL is parsed server-side; the embed iframe is rendered in a thin frame. FTS uses a stored generated `tsvector` column populated from title + excerpt + content in both locales, indexed with GIN.

**Tech Stack:** `next-mdx-remote` (RSC), `gray-matter` (not needed — DB-only), Drizzle, Neon, `@vercel/og` (Plan 6 covers OG), React.cache.

export type SpotifyKind = "track" | "album" | "playlist" | "episode";
export type Spotify = { kind: SpotifyKind; id: string };

const PATTERN = /^https:\/\/open\.spotify\.com\/(track|album|playlist|episode)\/([a-zA-Z0-9]{10,})(?:\?.*)?$/;

export function parseSpotifyUrl(url: string | null | undefined): Spotify | null {
  if (!url) return null;
  const m = PATTERN.exec(url);
  if (!m) return null;
  return { kind: m[1] as SpotifyKind, id: m[2]! };
}

export function toEmbedSrc(s: Spotify) {
  return `https://open.spotify.com/embed/${s.kind}/${s.id}`;
}
```

  ADD COLUMN IF NOT EXISTS "search_vector" tsvector

    setweight(to_tsvector('simple', coalesce(title_en, '')), 'A') ||

import "server-only";
import { cache } from "react";
import { and, desc, eq, isNull, or, sql } from "drizzle-orm";
import { db, schema } from "../client";

export type Locale = "tr" | "en";

const livePostsBase = () =>
  db
    .select()
    .from(schema.posts)
    .where(and(eq(schema.posts.status, "published"), isNull(schema.posts.deletedAt)));

export const listPublishedPosts = cache(
  async (opts: { limit?: number; offset?: number } = {}) => {
    const { limit = 20, offset = 0 } = opts;
    return livePostsBase()
      .orderBy(desc(schema.posts.publishedAt))
      .limit(limit)
      .offset(offset);
  },
);

export const countPublishedPosts = cache(async () => {
  const [row] = await db
    .select({ n: sql<number>`count(*)::int` })
    .from(schema.posts)
    .where(and(eq(schema.posts.status, "published"), isNull(schema.posts.deletedAt)));
  return row?.n ?? 0;
});

export const findPublishedBySlug = cache(async (locale: Locale, slug: string) => {
  const col = locale === "tr" ? schema.posts.slugTr : schema.posts.slugEn;
  const rows = await db
    .select()
    .from(schema.posts)
    .where(
      and(
        eq(col, slug),
        eq(schema.posts.status, "published"),
        isNull(schema.posts.deletedAt),
      ),
    )
    .limit(1);
  return rows[0] ?? null;
});

export const searchPublished = cache(async (query: string, locale: Locale) => {
  const q = query.trim();
  if (!q) return [];
  return db
    .select({
      id: schema.posts.id,
      slugTr: schema.posts.slugTr,
      slugEn: schema.posts.slugEn,
      titleTr: schema.posts.titleTr,
      titleEn: schema.posts.titleEn,
      excerptTr: schema.posts.excerptTr,
      excerptEn: schema.posts.excerptEn,
      publishedAt: schema.posts.publishedAt,
      rank: sql<number>`ts_rank(${schema.posts.searchVector}, plainto_tsquery('simple', ${q}))`,
    })
    .from(schema.posts)
    .where(
      and(
        eq(schema.posts.status, "published"),
        isNull(schema.posts.deletedAt),
        sql`${schema.posts.searchVector} @@ plainto_tsquery('simple', ${q})`,
      ),
    )
    .orderBy(desc(sql`ts_rank(${schema.posts.searchVector}, plainto_tsquery('simple', ${q}))`))
    .limit(50);
});

export const listAllPublishedSlugsForSitemap = cache(async () => {
  return db
    .select({
      slugTr: schema.posts.slugTr,
      slugEn: schema.posts.slugEn,
      updatedAt: schema.posts.updatedAt,
    })
    .from(schema.posts)
    .where(and(eq(schema.posts.status, "published"), isNull(schema.posts.deletedAt)));
});
```

`cache` ensures per-request dedup — repeated calls in the same Server Component render reuse the result.

### Task 4: Locale selector for posts

**Files:**

- [ ] **Step 1: Implement**

```ts
import type { Locale } from "@/lib/i18n/config";
import type { InferSelectModel } from "drizzle-orm";
import type { schema } from "@/lib/db/client";

type DbPost = InferSelectModel<typeof schema.posts>;

export type ResolvedPost = {
  id: string;
  locale: Locale;
  slug: string;
  title: string;
  excerpt: string | null;
  content: string;
  coverImageUrl: string | null;
  coverImageAlt: string | null;
  spotifyUrl: string | null;
  publishedAt: Date | null;
  hasOtherLocale: boolean;
  otherSlug: string | null;
};

export function resolveForLocale(post: DbPost, locale: Locale): ResolvedPost {
  const tr = {
    slug: post.slugTr,
    title: post.titleTr,
    excerpt: post.excerptTr,
    content: post.contentTr,
    alt: post.coverImageAltTr,
  };
  const en = {
    slug: post.slugEn,
    title: post.titleEn,
    excerpt: post.excerptEn,
    content: post.contentEn,
    alt: post.coverImageAltEn,
  };
  const current = locale === "tr" ? tr : en;
  const other = locale === "tr" ? en : tr;
  const otherComplete = Boolean(other.title && other.content);

  return {
    id: post.id,
    locale,
    slug: current.slug,
    title: current.title,
    excerpt: current.excerpt,
    content: current.content,
    coverImageUrl: post.coverImageUrl,
    coverImageAlt: current.alt,
    spotifyUrl: post.spotifyUrl,
    publishedAt: post.publishedAt,
    hasOtherLocale: otherComplete,
    otherSlug: otherComplete ? other.slug : null,
  };
}

export function isLocaleComplete(post: DbPost, locale: Locale): boolean {
  if (locale === "tr") return Boolean(post.titleTr && post.contentTr);
  return Boolean(post.titleEn && post.contentEn);
}
```

"use client";

import { useRouter, usePathname } from "@/lib/i18n/routing";
import { useState, useTransition } from "react";
import { useTranslations } from "next-intl";

export function SearchBox({ initialQuery = "" }: { initialQuery?: string }) {
  const router = useRouter();
  const pathname = usePathname();
  const t = useTranslations("common");
  const [q, setQ] = useState(initialQuery);
  const [pending, startTransition] = useTransition();

  function onSubmit(e: React.FormEvent) {
    e.preventDefault();

      const trimmed = q.trim();

    });
  }

  return (
    <form onSubmit={onSubmit} role="search" className="flex items-center gap-3 border-b border-rule pb-2">
      <input
        type="search"
        value={q}
        onChange={(e) => setQ(e.target.value)}
        placeholder="Search…"
        className="w-full bg-transparent text-small outline-none"
        aria-label="Search"
      />
      <button type="submit" disabled={pending} className="text-small uppercase tracking-wider text-fg-muted hover:text-fg">
        {pending ? t("loading") : "→"}
      </button>
    </form>
  );
}

```tsx
import { setRequestLocale, getTranslations } from "next-intl/server";
import { Container } from "@/components/public/Container";
import { BlogCard } from "@/components/blog/BlogCard";
import { Pagination } from "@/components/blog/Pagination";
import { SearchBox } from "@/components/blog/SearchBox";
import { resolveForLocale } from "@/lib/blog/post";
import {
  listPublishedPosts,
  countPublishedPosts,
  searchPublished,
} from "@/lib/db/queries/posts";
import { type Locale } from "@/lib/i18n/config";

export const revalidate = 60;

const PAGE_SIZE = 12;

export default async function BlogIndex({
  params,
  searchParams,
}: {
  params: Promise<{ locale: Locale }>;
  searchParams: Promise<{ q?: string; page?: string }>;
}) {
  const { locale } = await params;
  const { q, page: pageRaw } = await searchParams;
  setRequestLocale(locale);
  const t = await getTranslations();

  const page = Math.max(1, Number.parseInt(pageRaw ?? "1", 10) || 1);

  if (q && q.trim()) {
    const results = await searchPublished(q, locale);
    const resolved = results
      .filter((r) => (locale === "tr" ? r.titleTr : r.titleEn))
      .map((r) => ({
        slug: locale === "tr" ? r.slugTr : r.slugEn,
        title: (locale === "tr" ? r.titleTr : r.titleEn) ?? "",
        excerpt: (locale === "tr" ? r.excerptTr : r.excerptEn) ?? null,
        date: r.publishedAt,
      }));
    return (
      <section className="pt-24 md:pt-40">
        <Container width="editorial">
          <h1 className="font-serif text-display">{t("nav.blog")}</h1>
          <div className="mt-12">
            <SearchBox initialQuery={q} />
          </div>
          {resolved.length === 0 ? (
            <p className="mt-16 text-fg-muted">{t("common.noResults")}</p>
          ) : (
            <ul className="mt-16 space-y-12">
              {resolved.map((r) => (
                <li key={r.slug}>
                  <BlogCard
                    slug={r.slug}
                    title={r.title}
                    excerpt={r.excerpt}
                    coverImageUrl={null}
                    coverImageAlt={null}
                    date={r.date}
                  />
                </li>
              ))}
            </ul>
          )}
        </Container>
      </section>
    );
  }

  const [postsAll, total] = await Promise.all([
    listPublishedPosts({ limit: PAGE_SIZE, offset: (page - 1) * PAGE_SIZE }),
    countPublishedPosts(),
  ]);
  const resolved = postsAll
    .map((p) => resolveForLocale(p, locale))
    .filter((p) => p.title && p.content);
  const totalPages = Math.max(1, Math.ceil(total / PAGE_SIZE));

  return (
    <section className="pt-24 md:pt-40">
      <Container width="gallery">
        <h1 className="font-serif text-display">{t("nav.blog")}</h1>
        <div className="mt-12 max-w-prose">
          <SearchBox />
        </div>
        {resolved.length === 0 ? (
          <p className="mt-16 text-fg-muted">—</p>
        ) : (
          <div className="mt-16 grid grid-cols-1 gap-12 md:grid-cols-2 lg:grid-cols-3">
            {resolved.map((p) => (
              <BlogCard
                key={p.id}
                slug={p.slug}
                title={p.title}
                excerpt={p.excerpt}
                coverImageUrl={p.coverImageUrl}
                coverImageAlt={p.coverImageAlt}
                date={p.publishedAt}
              />
            ))}
          </div>
        )}
        <Pagination page={page} totalPages={totalPages} />
      </Container>
    </section>
  );
}
```

`listPublishedPosts` and `countPublishedPosts` run in parallel via `Promise.all` (per `async-parallel`).

### Task 10: Blog detail page

**Files:**
- Create: `app/[locale]/(public)/blog/[slug]/page.tsx`

- [ ] **Step 1: Implement**

```tsx
import { notFound } from "next/navigation";
import { MDXRemote } from "next-mdx-remote/rsc";
import remarkGfm from "remark-gfm";
import { setRequestLocale } from "next-intl/server";
import { Container } from "@/components/public/Container";
import { PostHeader } from "@/components/blog/PostHeader";
import { SpotifyEmbed } from "@/components/blog/SpotifyEmbed";
import { MissingTranslation } from "@/components/blog/MissingTranslation";
import { mdxComponents } from "@/lib/mdx/components";
import { findPublishedBySlug, listPublishedPosts } from "@/lib/db/queries/posts";
import { resolveForLocale, isLocaleComplete } from "@/lib/blog/post";
import { type Locale, locales } from "@/lib/i18n/config";

export const revalidate = 60;
export const dynamicParams = true;

export async function generateStaticParams() {
  const all = await listPublishedPosts({ limit: 200 });
  const out: { locale: Locale; slug: string }[] = [];
  for (const p of all) {
    if (p.slugTr && p.titleTr && p.contentTr) out.push({ locale: "tr", slug: p.slugTr });
    if (p.slugEn && p.titleEn && p.contentEn) out.push({ locale: "en", slug: p.slugEn });
  }
  return out;
}

export default async function BlogDetail({
  params,
}: {
  params: Promise<{ locale: Locale; slug: string }>;
}) {
  const { locale, slug } = await params;
  if (!locales.includes(locale)) notFound();
  setRequestLocale(locale);

  const post = await findPublishedBySlug(locale, slug);
  if (!post) {
    // Try the other locale — if the slug exists there we show missing-translation
    const otherLocale: Locale = locale === "tr" ? "en" : "tr";
    const otherPost = await findPublishedBySlug(otherLocale, slug);
    if (!otherPost) notFound();
    if (!isLocaleComplete(otherPost, locale)) {
      const resolved = resolveForLocale(otherPost, otherLocale);
      return (
        <MissingTranslation
          currentLocale={locale}
          otherSlug={resolved.slug}
          otherTitle={resolved.title}
        />
      );
    }
    notFound();
  }

  if (!isLocaleComplete(post, locale)) {
    const otherLocale: Locale = locale === "tr" ? "en" : "tr";
    const resolved = resolveForLocale(post, otherLocale);
    return (
      <MissingTranslation
        currentLocale={locale}
        otherSlug={resolved.slug}
        otherTitle={resolved.title}
      />
    );
  }

  const resolved = resolveForLocale(post, locale);

  return (
    <article>
      <PostHeader
        title={resolved.title}
        publishedAt={resolved.publishedAt}
        coverImageUrl={resolved.coverImageUrl}
        coverImageAlt={resolved.coverImageAlt}
      />

      <Container width="prose" className="py-16">
        {resolved.spotifyUrl ? <SpotifyEmbed url={resolved.spotifyUrl} /> : null}
        <div className="prose-editorial">
          <MDXRemote
            source={resolved.content}
            components={mdxComponents}
            options={{
              mdxOptions: {
                remarkPlugins: [remarkGfm],
                rehypePlugins: [],
              },
              parseFrontmatter: false,
            }}
          />
        </div>
      </Container>
    </article>
  );
}
```
