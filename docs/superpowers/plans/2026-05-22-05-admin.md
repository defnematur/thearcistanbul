# Admin Interface Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** English-only admin at `/admin/*` for the two seeded editors. Login → dashboard → post list → create/edit form with locale tabs, MDX editor, cover image upload (Vercel Blob), Spotify URL field with live embed preview, draft/publish flow.

**Architecture:** All admin pages live under `app/admin` outside the `[locale]` segment. Middleware (from Plan 2) gates everything except `/admin/login`. Forms call Server Actions which validate with Zod, write via Drizzle, and call `revalidatePath()` to refresh public pages. The Markdown editor is loaded via `next/dynamic({ ssr: false })` (heavy client lib — `bundle-dynamic-imports`). Image upload goes through a Server Action that streams to Vercel Blob and strips EXIF with sharp.

**Tech Stack:** Auth.js v5 (from Plan 2), Drizzle (Plan 2), `@vercel/blob`, `sharp`, `@uiw/react-md-editor` (loaded dynamically), `react-hook-form`, Zod, shadcn/ui primitives (Button, Input, Tabs).

```tsx
"use client";

import { useId, useState, type ReactNode } from "react";
import { clsx } from "clsx";

type Tab = { value: string; label: string; content: ReactNode };

export function Tabs({ tabs, defaultValue }: { tabs: Tab[]; defaultValue?: string }) {
  const idBase = useId();
  const [active, setActive] = useState(defaultValue ?? tabs[0]?.value);
  if (!active) return null;
  return (
    <div>
      <div role="tablist" className="flex gap-6 border-b border-rule">
        {tabs.map((t) => (
          <button
            key={t.value}
            id={`${idBase}-tab-${t.value}`}
            role="tab"
            aria-selected={active === t.value}
            aria-controls={`${idBase}-panel-${t.value}`}
            onClick={() => setActive(t.value)}
            className={clsx(
              "border-b py-3 text-small uppercase tracking-wider transition-colors",
              active === t.value ? "border-fg text-fg" : "border-transparent text-fg-muted hover:text-fg",
            )}
          >
            {t.label}
          </button>
        ))}
      </div>
      {tabs.map((t) => (
        <div
          key={t.value}
          id={`${idBase}-panel-${t.value}`}
          role="tabpanel"
          aria-labelledby={`${idBase}-tab-${t.value}`}
          hidden={active !== t.value}
          className="pt-8"
        >
          {t.content}
        </div>
      ))}
    </div>
  );
}
```

```ts
const TR_MAP: Record<string, string> = {
  ş: "s", Ş: "s",
  ı: "i", İ: "i",
  ğ: "g", Ğ: "g",
  ü: "u", Ü: "u",
  ö: "o", Ö: "o",
  ç: "c", Ç: "c",
};

export function slugify(input: string): string {
  if (!input) return "";
  let s = input.trim();
  s = s.replace(/[şŞıİğĞüÜöÖçÇ]/g, (ch) => TR_MAP[ch] ?? ch);
  s = s.normalize("NFKD").replace(/\p{Diacritic}/gu, "");
  s = s.toLowerCase();
  s = s.replace(/[^a-z0-9]+/g, "-");
  s = s.replace(/^-+|-+$/g, "");
  return s;
}
```

```ts
import { z } from "zod";

const SPOTIFY = /^https:\/\/open\.spotify\.com\/(track|album|playlist|episode)\/[a-zA-Z0-9]{10,}(\?.*)?$/;
const SLUG = /^[a-z0-9]+(-[a-z0-9]+)*$/;

const localeFields = z.object({
  slug: z.string().regex(SLUG, "invalid_slug").max(200),
  title: z.string().min(1, "required").max(200),
  excerpt: z.string().max(300).optional().default(""),
  content: z.string().max(200_000).optional().default(""),
  coverImageAlt: z.string().max(250).optional().default(""),
});

export const postSaveSchema = z.object({
  id: z.string().uuid().optional(),
  tr: localeFields,
  en: localeFields,
  coverImageUrl: z.string().url().nullable(),
  spotifyUrl: z
    .string()
    .regex(SPOTIFY, "invalid_spotify_url")
    .nullable()
    .or(z.literal("").transform(() => null)),
  status: z.enum(["draft", "published"]),
  publishedAt: z.string().datetime().nullable(),
  updatedAt: z.string().datetime().nullable(), // optimistic concurrency
});

export type PostSaveInput = z.infer<typeof postSaveSchema>;

export function readyToPublish(input: PostSaveInput): boolean {
  const trOk = !!input.tr.title && !!input.tr.content;
  const enOk = !!input.en.title && !!input.en.content;
  return trOk && enOk;
}
```

```ts
  const rows = await db
    .select()
    .from(schema.posts)
    .where(and(eq(schema.posts.id, id), isNull(schema.posts.deletedAt)))

  return rows[0] ?? null;

export async function slugInUse(
  locale: "tr" | "en",
  slug: string,
  excludeId?: string,
) {
  const col = locale === "tr" ? schema.posts.slugTr : schema.posts.slugEn;
  const rows = await db
    .select({ id: schema.posts.id })
    .from(schema.posts)
    .where(
      and(
        eq(col, slug),
        isNull(schema.posts.deletedAt),
        excludeId ? ne(schema.posts.id, excludeId) : undefined,
      ),
    )
    .limit(1);
  return rows.length > 0;
}
```

```ts
"use server";

import { put } from "@vercel/blob";
import sharp from "sharp";
import { randomUUID } from "node:crypto";
import { auth } from "@/lib/auth";

const ALLOWED = new Set(["image/jpeg", "image/png", "image/webp", "image/avif"]);
const MAX_BYTES = 8 * 1024 * 1024;

type UploadResult = { ok: true; url: string } | { ok: false; error: string };

export async function uploadCoverImage(formData: FormData): Promise<UploadResult> {
  const session = await auth();
  if (!session?.user) return { ok: false, error: "unauthenticated" };

  const file = formData.get("file");
  if (!(file instanceof File)) return { ok: false, error: "no_file" };
  if (!ALLOWED.has(file.type)) return { ok: false, error: "bad_type" };
  if (file.size > MAX_BYTES) return { ok: false, error: "too_large" };

  const bytes = Buffer.from(await file.arrayBuffer());
  const processed = await sharp(bytes)
    .rotate()
    .withMetadata({ exif: {} })
    .toBuffer();

  const ext =
    file.type === "image/png" ? "png" :
    file.type === "image/webp" ? "webp" :
    file.type === "image/avif" ? "avif" : "jpg";

  const name = `covers/${randomUUID()}.${ext}`;
  const blob = await put(name, processed, { access: "public", contentType: file.type });
  return { ok: true, url: blob.url };
}
```

- [ ] **Step 1: Implement**

```ts
"use server";

import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";
import { and, eq } from "drizzle-orm";
import { db, schema } from "@/lib/db/client";
import { auth } from "@/lib/auth";
import { postSaveSchema, readyToPublish, type PostSaveInput } from "@/lib/validators/post";
import { slugInUse } from "@/lib/db/queries/posts-admin";

type ActionResult =
  | { ok: true; id: string }
  | { ok: false; error: string; fieldErrors?: Record<string, string> };

async function requireUser() {
  const session = await auth();
  if (!session?.user?.id) throw new Error("unauthenticated");
  return session.user;
}

export async function savePost(raw: PostSaveInput, action: "save" | "publish"): Promise<ActionResult> {
  const user = await requireUser();
  const parsed = postSaveSchema.safeParse(raw);
  if (!parsed.success) {
    return { ok: false, error: "validation" };
  }
  const data = parsed.data;

  if (action === "publish" && !readyToPublish(data)) {
    return { ok: false, error: "missing_translation" };
  }

  for (const loc of ["tr", "en"] as const) {
    if (await slugInUse(loc, data[loc].slug, data.id)) {
      return { ok: false, error: "slug_taken", fieldErrors: { [`${loc}.slug`]: "slug_taken" } };
    }
  }

  const baseRow = {
    slugTr: data.tr.slug,
    slugEn: data.en.slug,
    titleTr: data.tr.title,
    titleEn: data.en.title,
    excerptTr: data.tr.excerpt || null,
    excerptEn: data.en.excerpt || null,
    contentTr: data.tr.content || "",
    contentEn: data.en.content || "",
    coverImageUrl: data.coverImageUrl ?? null,
    coverImageAltTr: data.tr.coverImageAlt || null,
    coverImageAltEn: data.en.coverImageAlt || null,
    spotifyUrl: data.spotifyUrl ?? null,
    status: action === "publish" ? ("published" as const) : data.status,
    authorId: user.id,
    updatedAt: new Date(),
  };

  let id = data.id;

  if (!id) {
    const [row] = await db
      .insert(schema.posts)
      .values({
        ...baseRow,
        publishedAt: action === "publish" ? new Date() : data.publishedAt ? new Date(data.publishedAt) : null,
      })
      .returning({ id: schema.posts.id });
    if (!row) return { ok: false, error: "insert_failed" };
    id = row.id;
  } else {
    if (data.updatedAt) {
      const stamp = new Date(data.updatedAt);
      const current = await db
        .select({ updatedAt: schema.posts.updatedAt })
        .from(schema.posts)
        .where(eq(schema.posts.id, id))
        .limit(1);
      if (current[0] && current[0].updatedAt.getTime() !== stamp.getTime()) {
        return { ok: false, error: "stale" };
      }
    }
    const publishUpdate =
      action === "publish"
        ? { publishedAt: new Date() }
        : data.publishedAt
          ? { publishedAt: new Date(data.publishedAt) }
          : {};
    await db
      .update(schema.posts)
      .set({ ...baseRow, ...publishUpdate })
      .where(eq(schema.posts.id, id));
  }

  revalidatePath("/[locale]", "layout");
  revalidatePath("/[locale]/blog", "page");
  revalidatePath(`/tr/blog/${data.tr.slug}`, "page");
  revalidatePath(`/en/blog/${data.en.slug}`, "page");

  return { ok: true, id };
}

export async function deletePost(id: string) {
  await requireUser();
  await db
    .update(schema.posts)
    .set({ deletedAt: new Date() })
    .where(eq(schema.posts.id, id));
  revalidatePath("/[locale]/blog", "page");
  redirect("/admin/posts");
}
```

```ts
"use server";

import { signIn, signOut } from "@/lib/auth";
import { headers } from "next/headers";
import { redirect } from "next/navigation";
import { AuthError } from "next-auth";

export async function loginAction(formData: FormData) {
  const email = String(formData.get("email") ?? "");
  const password = String(formData.get("password") ?? "");
  const from = String(formData.get("from") ?? "/admin");
  const hdrs = await headers();
  const ip =
    hdrs.get("x-forwarded-for")?.split(",")[0]?.trim() ??
    hdrs.get("x-real-ip") ??
    "unknown";

  try {
    await signIn("credentials", {
      email,
      password,
      ip,
      redirect: false,
    });
  } catch (e) {
    if (e instanceof AuthError) return { ok: false as const, error: "invalid" };
    throw e;
  }
  redirect(from);
}

export async function logoutAction() {
  await signOut({ redirect: false });
  redirect("/admin/login");
}
```

```tsx
"use client";

import { useState, useTransition } from "react";

import { loginAction } from "@/lib/actions/auth";
```

```tsx
"use client";

import { useState, useTransition } from "react";
import Image from "next/image";
import { uploadCoverImage } from "@/lib/actions/uploads";
import { Button } from "@/components/ui/Button";

export function CoverImageUploader({
  value,
  onChange,
}: {
  value: string | null;
  onChange: (url: string | null) => void;
}) {
  const [error, setError] = useState<string | null>(null);
  const [pending, startTransition] = useTransition();

  function onFileChange(e: React.ChangeEvent<HTMLInputElement>) {
    const file = e.target.files?.[0];
    if (!file) return;
    setError(null);
    const fd = new FormData();
    fd.append("file", file);
    startTransition(async () => {
      const res = await uploadCoverImage(fd);
      if (res.ok) onChange(res.url);
      else setError(res.error);
    });
  }

  return (
    <div>
      {value ? (
        <div className="relative aspect-[3/2] w-full max-w-md overflow-hidden border border-rule">
          <Image src={value} alt="" fill sizes="600px" className="object-cover" />
        </div>
      ) : (
        <div className="flex aspect-[3/2] w-full max-w-md items-center justify-center border border-dashed border-rule text-small text-fg-muted">
          No cover image
        </div>
      )}
      <div className="mt-4 flex items-center gap-4">
        <input
          type="file"
          accept="image/jpeg,image/png,image/webp,image/avif"
          onChange={onFileChange}
          disabled={pending}
          className="text-small"
        />
        {value ? (
          <Button type="button" variant="ghost" onClick={() => onChange(null)}>
            Remove
          </Button>
        ) : null}
      </div>
      {pending ? <p className="mt-2 text-small text-fg-muted">Uploading…</p> : null}
      {error ? <p className="mt-2 text-small text-danger">{error}</p> : null}
    </div>
  );
}
```

```tsx
"use client";

import { useState, useTransition } from "react";
import { useRouter } from "next/navigation";
import { useForm, Controller } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { Button } from "@/components/ui/Button";
import { Input } from "@/components/ui/Input";
import { Textarea } from "@/components/ui/Textarea";
import { Label } from "@/components/ui/Label";
import { Tabs } from "@/components/ui/Tabs";
import { MarkdownEditor } from "./MarkdownEditor";
import { CoverImageUploader } from "./CoverImageUploader";
import { SpotifyPreview } from "./SpotifyPreview";
import { postSaveSchema, type PostSaveInput } from "@/lib/validators/post";
import { slugify } from "@/lib/slugify";
import { savePost, deletePost } from "@/lib/actions/posts";

export function PostForm({ initial }: { initial: PostSaveInput }) {
  const router = useRouter();
  const [pending, startTransition] = useTransition();
  const [topError, setTopError] = useState<string | null>(null);

  const form = useForm<PostSaveInput>({
    resolver: zodResolver(postSaveSchema),
    defaultValues: initial,
  });

  function submit(action: "save" | "publish") {
    setTopError(null);
    form.handleSubmit((data) => {
      startTransition(async () => {
        const res = await savePost(data, action);
        if (res.ok) {
          if (!initial.id) router.replace(`/admin/posts/${res.id}/edit`);
          else router.refresh();
        } else {
          setTopError(res.error);
        }
      });
    })();
  }

  function autoSlug(locale: "tr" | "en") {
    const title = form.getValues(`${locale}.title`);
    if (title && !form.getValues(`${locale}.slug`)) {
      form.setValue(`${locale}.slug`, slugify(title));
    }
  }

  function localePanel(loc: "tr" | "en") {
    const errs = form.formState.errors[loc] ?? {};
    return (
      <div className="grid gap-6">
        <div>
          <Label htmlFor={`${loc}-title`}>Title</Label>
          <Input id={`${loc}-title`} {...form.register(`${loc}.title`)} onBlur={() => autoSlug(loc)} className="mt-2" />
          {errs.title ? <p className="mt-2 text-small text-danger">{errs.title.message}</p> : null}
        </div>
        <div>
          <Label htmlFor={`${loc}-slug`}>Slug</Label>
          <Input id={`${loc}-slug`} {...form.register(`${loc}.slug`)} className="mt-2" />
          {errs.slug ? <p className="mt-2 text-small text-danger">{errs.slug.message}</p> : null}
        </div>
        <div>
          <Label htmlFor={`${loc}-excerpt`}>Excerpt</Label>
          <Textarea id={`${loc}-excerpt`} rows={3} {...form.register(`${loc}.excerpt`)} className="mt-2" />
        </div>
        <div>
          <Label htmlFor={`${loc}-alt`}>Cover image alt</Label>
          <Input id={`${loc}-alt`} {...form.register(`${loc}.coverImageAlt`)} className="mt-2" />
        </div>
        <div>
          <Label>Content (Markdown)</Label>
          <Controller
            control={form.control}
            name={`${loc}.content`}
            render={({ field }) => (
              <div className="mt-2">
                <MarkdownEditor value={field.value ?? ""} onChange={field.onChange} />
              </div>
            )}
          />
        </div>
      </div>
    );
  }

  const spotifyUrl = form.watch("spotifyUrl") ?? "";

  return (
    <form className="grid gap-12">
      <Tabs
        tabs={[
          { value: "tr", label: "Turkish", content: localePanel("tr") },
          { value: "en", label: "English", content: localePanel("en") },
        ]}
      />

      <section>
        <h2 className="font-serif text-h2">Cover image</h2>
        <Controller
          control={form.control}
          name="coverImageUrl"
          render={({ field }) => (
            <div className="mt-4">
              <CoverImageUploader value={field.value} onChange={field.onChange} />
            </div>
          )}
        />
      </section>

      <section>
        <h2 className="font-serif text-h2">Spotify</h2>
        <Controller
          control={form.control}
          name="spotifyUrl"
          render={({ field }) => (
            <div className="mt-4">
              <Input
                value={field.value ?? ""}
                onChange={(e) => field.onChange(e.target.value || null)}
                placeholder="https://open.spotify.com/track/…"
              />
              <SpotifyPreview url={spotifyUrl} />
            </div>
          )}
        />
      </section>

      <section className="border-t border-rule pt-8">
        {topError ? <p className="mb-4 text-small text-danger">Error: {topError}</p> : null}
        <div className="flex flex-wrap items-center gap-4">
          <Button type="button" variant="secondary" onClick={() => submit("save")} disabled={pending}>
            Save draft
          </Button>
          <Button type="button" onClick={() => submit("publish")} disabled={pending}>
            Publish
          </Button>
          {initial.id ? (
            <Button
              type="button"
              variant="danger"
              onClick={() => {
                if (!confirm("Delete this post?")) return;
                startTransition(() => deletePost(initial.id!));
              }}
              disabled={pending}
            >
              Delete
            </Button>
          ) : null}
        </div>
      </section>
    </form>
  );
}
```
